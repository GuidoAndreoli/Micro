.include "m328pdef.inc"

.org 0x0000
    rjmp RESET

RESET:
    ; Inicializar Stack Pointer
    ldi r16, HIGH(RAMEND)
    out SPH, r16
    ldi r16, LOW(RAMEND)
    out SPL, r16

    ; Configurar PB3 (OC0A) como salida
    ldi r16, (1 << DDB3)
    out DDRB, r16

    ; Configurar Timer0 en modo Fast PWM, no inversor
    ldi r16, (1 << COM0A1) | (1 << WGM01) | (1 << WGM00)
    out TCCR0A, r16

    ; Prescaler = 64 -> f_PWM = 16MHz / (64 * 256) ≈ 976 Hz
    ldi r16, (1 << CS01) | (1 << CS00)
    out TCCR0B, r16

    ; Duty Cycle (OCR0A = 128 → 50%)
    ldi r16, 128
    out OCR0A, r16

LOOP:
    rjmp LOOP
